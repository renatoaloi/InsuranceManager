```csharp
public bool CanTransitionTo(ProposalStatus targetStatus)
{
    return (Status, targetStatus) switch
    {
        (ProposalStatus.EmAnalise, ProposalStatus.Aprovada) => true,
        (ProposalStatus.EmAnalise, ProposalStatus.Recusada) => true,
        _ => false
    };
}
```
