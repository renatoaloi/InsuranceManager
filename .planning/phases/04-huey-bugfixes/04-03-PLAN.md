    ```
    # Note the proposal ID from response

    # Trigger status change (this enqueues Huey task)
    curl -X POST http://localhost:5000/api/proposals/{id}/status \

      -H "Content-Type: application/json" \
      -d '{"newStatus":"Aprovada"}'
    ```
